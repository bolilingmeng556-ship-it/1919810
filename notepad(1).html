<script type="importmap">
{"imports":{
"three":"https://unpkg.com/three@0.126.0/build/three.module.js",
"three/addons/":"https://unpkg.com/three@0.126.0/examples/jsm/",
"cannon-es":"https://unpkg.com/cannon-es@0.20.0/dist/cannon-es.js"
}}

<script type="module">
import*as THREE from'three';
import{OrbitControls}from'three/addons/controls/OrbitControls.js';
import{GLTFLoader}from'three/addons/loaders/GLTFLoader.js';
import{OBJLoader}from'three/addons/loaders/OBJLoader.js';
import{FBXLoader}from'three/addons/loaders/FBXLoader.js';
import*as CANNON from'cannon-es';

// 场景
const scene=new THREE.Scene();
scene.background=new THREE.Color(0x111122);
const camera=new THREE.PerspectiveCamera(45,innerWidth/innerHeight,.1,1000);
camera.position.set(4,4,8);
const renderer=new THREE.WebGLRenderer({antialias:true});
renderer.setSize(innerWidth,innerHeight);
renderer.setPixelRatio(Math.min(devicePixelRatio,2));
// ===== 开启阴影 =====
renderer.shadowMap.enabled=true;
renderer.shadowMap.type=THREE.PCFSoftShadowMap;
document.body.appendChild(renderer.domElement);

// 控制器
const controls=new OrbitControls(camera,renderer.domElement);
controls.enableDamping=true;
controls.target.set(0,1,0);

// 灯光
scene.add(new THREE.AmbientLight(0xffffff,.5));
const dl=new THREE.DirectionalLight(0xffffff,1.8);
dl.position.set(5,10,7);
// ===== 主光源投射阴影 =====
dl.castShadow=true;
dl.shadow.mapSize.width=1024;
dl.shadow.mapSize.height=1024;
const d=12;
dl.shadow.camera.left=-d;
dl.shadow.camera.right=d;
dl.shadow.camera.top=d;
dl.shadow.camera.bottom=-d;
dl.shadow.camera.near=1;
dl.shadow.camera.far=30;
dl.shadow.bias=-0.0005;
scene.add(dl);
// 补光，不投影
const back=new THREE.PointLight(0x4466ff,.5);
back.position.set(-5,3,-5);
scene.add(back);

// 地面
const ground=new THREE.Mesh(
  new THREE.CircleGeometry(20,32),
  new THREE.MeshStandardMaterial({color:0x2a2a3a,roughness:.8,metalness:.1,side:THREE.DoubleSide})
);
ground.rotation.x=-Math.PI/2;
// ===== 地面接收阴影 =====
ground.receiveShadow=true;
scene.add(ground);
const grid=new THREE.GridHelper(30,20,0x88aaff,0x445588);
grid.position.y=0.01;
scene.add(grid);

// 物理世界
const world=new CANNON.World();
world.gravity.set(0,-9.82,0);
world.broadphase=new CANNON.SAPBroadphase(world);
world.defaultContactMaterial.friction=0.4;
world.defaultContactMaterial.restitution=0.2;
const groundBody=new CANNON.Body({mass:0,shape:new CANNON.Plane()});
groundBody.quaternion.setFromAxisAngle(new CANNON.Vec3(1,0,0),-Math.PI/2);
world.addBody(groundBody);

let bodies=[];

// 加载模型
document.getElementById('file').onchange=e=>{
  const f=e.target.files[0];if(!f)return;
  clearAll();
  const ext=f.name.split('.').pop().toLowerCase();
  let loader;
  if(ext==='glb'||ext==='gltf')loader=new GLTFLoader();
  else if(ext==='obj')loader=new OBJLoader();
  else if(ext==='fbx')loader=new FBXLoader();
  else{return alert('请用 GLB/GLTF/OBJ/FBX')}
 
  const r=new FileReader();
  r.onload=ev=>{
    loader.parse(ev.target.result,'',res=>{
      const model=ext==='glb'||ext==='gltf'?res.scene:res;
      // 位置调整
      const box=new THREE.Box3().setFromObject(model);
      const c=box.getCenter(new THREE.Vector3());
      model.position.y-=box.min.y;
      model.position.x-=c.x;
      model.position.z-=c.z;
      scene.add(model);
      model.updateMatrixWorld(true);
     
      // 拆解mesh
      const meshes=[];
      model.traverse(o=>{if(o.isMesh)meshes.push(o)});
      if(!meshes.length){alert('模型没有网格');scene.remove(model);return}
     
      const data=meshes.map(m=>{
        m.updateWorldMatrix(true,false);
        const p=new THREE.Vector3(),q=new THREE.Quaternion(),s=new THREE.Vector3();
        m.matrixWorld.decompose(p,q,s);
        return{m,p,q,s};
      });
     
      data.forEach(d=>{
        if(d.m.parent)d.m.parent.remove(d.m);
        scene.add(d.m);
        d.m.position.copy(d.p);d.m.quaternion.copy(d.q);d.m.scale.copy(d.s);
        // ===== 部件投射和接收阴影 =====
        d.m.castShadow=true;
        d.m.receiveShadow=true;
        // 材质补充：如果是标准材质，可开启阴影相关（一般默认即可）
      });
      scene.remove(model);
     
      // 创建物理体
      data.forEach(d=>{
        const m=d.m;
        if(!m.geometry.boundingBox)m.geometry.computeBoundingBox();
        const bb=m.geometry.boundingBox;
        const size=new THREE.Vector3().subVectors(bb.max,bb.min);
        const hx=Math.max(size.x*m.scale.x/2,.1);
        const hy=Math.max(size.y*m.scale.y/2,.1);
        const hz=Math.max(size.z*m.scale.z/2,.1);
        const gc=new THREE.Vector3().addVectors(bb.min,bb.max).multiplyScalar(.5);
        const wc=gc.clone().applyMatrix4(m.matrixWorld);
        const body=new CANNON.Body({mass:1});
        body.addShape(new CANNON.Box(new CANNON.Vec3(hx,hy,hz)));
        body.position.set(wc.x,wc.y,wc.z);
        body.quaternion.set(m.quaternion.x,m.quaternion.y,m.quaternion.z,m.quaternion.w);
        body.angularDamping=.2;body.linearDamping=.1;
        world.addBody(body);
        bodies.push({body,mesh:m,offset:gc});
      });
      document.getElementById('tip').innerHTML='✅ 已导入<br>拖动视角观察倒下';
    },err=>{alert('加载失败:'+err.message)});
  };
  r.readAsArrayBuffer(f);
  e.target.value='';
};

document.getElementById('reset').onclick=()=>{clearAll();document.getElementById('tip').innerHTML='🧸 已重置<br>重新导入模型'};

function clearAll(){
  bodies.forEach(b=>{
    world.removeBody(b.body);
    if(b.mesh.parent)b.mesh.parent.remove(b.mesh);
  });
  bodies=[];
}

// 循环
function loop(){
  world.step(1/60,Math.min(.1,0.016),2);
  const q=new THREE.Quaternion();
  bodies.forEach(b=>{
    q.set(b.body.quaternion.x,b.body.quaternion.y,b.body.quaternion.z,b.body.quaternion.w);
    const off=b.offset.clone().applyQuaternion(q);
    b.mesh.position.copy(b.body.position).sub(off);
    b.mesh.quaternion.copy(q);
  });
  controls.update();
  renderer.render(scene,camera);
  requestAnimationFrame(loop);
}
loop();

addEventListener('resize',()=>{
  camera.aspect=innerWidth/innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth,innerHeight);
});
</script>
